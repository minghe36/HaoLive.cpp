cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" HAOLIVE_VERSION LIMIT_COUNT 1)
if(NOT HAOLIVE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain MAJOR.MINOR.PATCH")
endif()
project(HaoLive VERSION ${HAOLIVE_VERSION} LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8 /bigobj)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(HAOLIVE_METAL "Build Apple Metal backend" ${APPLE})
option(HAOLIVE_VULKAN "Build Vulkan backend (requires Vulkan SDK)" OFF)
option(HAOLIVE_CUDA "Build CUDA backend (requires CUDA toolkit)" OFF)
option(HAOLIVE_BUILD_TESTS "Build unit tests" ON)
option(HAOLIVE_BUILD_SHARED "Build optional shared library for C ABI/FFI callers" OFF)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependency libraries" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "Embed backends in the executable" FORCE)
set(GGML_METAL ${HAOLIVE_METAL} CACHE BOOL "" FORCE)
set(GGML_VULKAN ${HAOLIVE_VULKAN} CACHE BOOL "" FORCE)
set(GGML_CUDA ${HAOLIVE_CUDA} CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable binary")
set(GGML_OPENMP OFF CACHE BOOL "No OpenMP runtime")
set(GGML_BLAS OFF CACHE BOOL "CPU and requested GPU backends only" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/ggml EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
set(USE_TLS OFF CACHE BOOL "Loopback WebSocket uses owned local authentication" FORCE)
set(USE_ZLIB OFF CACHE BOOL "PCM messages do not use compression" FORCE)
set(BUILD_DEMO OFF CACHE BOOL "No dependency demo binaries" FORCE)
add_subdirectory(third_party/ixwebsocket EXCLUDE_FROM_ALL)

add_library(haolive-objects OBJECT src/runtime.cpp src/audio.cpp src/vad.cpp src/c_api.cpp src/models/xasr.cpp src/models/qwen.cpp src/models/moonshine.cpp src/models/nemotron.cpp)
set_target_properties(haolive-objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(haolive-objects PRIVATE include src)
target_link_libraries(haolive-objects PRIVATE ggml Threads::Threads)
target_compile_options(haolive-objects PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)
add_library(haolive STATIC $<TARGET_OBJECTS:haolive-objects>)
target_include_directories(haolive PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src)
target_link_libraries(haolive PRIVATE ggml Threads::Threads)
if(HAOLIVE_BUILD_SHARED)
    add_library(haolive-shared SHARED $<TARGET_OBJECTS:haolive-objects>)
    target_include_directories(haolive-shared PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
    target_link_libraries(haolive-shared PRIVATE ggml Threads::Threads)
    set_target_properties(haolive-shared PROPERTIES OUTPUT_NAME haolive VERSION ${PROJECT_VERSION} SOVERSION 1 WINDOWS_EXPORT_ALL_SYMBOLS ON)
    if(WIN32)
        set_target_properties(haolive-shared PROPERTIES ARCHIVE_OUTPUT_NAME haolive-ffi)
    endif()
endif()

add_executable(haolive-server src/server.cpp)
set(HAOLIVE_LICENSES "")
foreach(notice LICENSE THIRD_PARTY_NOTICES.md third_party/ggml/LICENSE
        third_party/ixwebsocket/LICENSE third_party/licenses/httplib.txt third_party/licenses/json.txt third_party/licenses/moonshine.txt)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${notice}" notice_text)
    string(APPEND HAOLIVE_LICENSES "\n${notice}\n\n${notice_text}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${notice}")
endforeach()
configure_file(cmake/licenses.hpp.in generated/licenses.hpp @ONLY)
configure_file(cmake/version.hpp.in generated/version.hpp @ONLY)
target_include_directories(haolive-server PRIVATE third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(haolive-server PRIVATE haolive ixwebsocket Threads::Threads)
set_target_properties(haolive-server PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

if(HAOLIVE_BUILD_TESTS)
    enable_testing()
    add_executable(haolive-tests tests/unit.cpp)
    target_include_directories(haolive-tests PRIVATE src)
    target_link_libraries(haolive-tests PRIVATE haolive ggml)
    add_test(NAME unit COMMAND haolive-tests)
    add_executable(haolive-xasr-streaming-tests tests/xasr_streaming.cpp)
    target_link_libraries(haolive-xasr-streaming-tests PRIVATE haolive)
    add_executable(haolive-qwen-streaming-tests tests/qwen_streaming.cpp)
    target_link_libraries(haolive-qwen-streaming-tests PRIVATE haolive)
    add_executable(haolive-vad-tests tests/vad_model.cpp)
    target_link_libraries(haolive-vad-tests PRIVATE haolive Threads::Threads)
    add_test(NAME version COMMAND haolive-server --version)
    set_tests_properties(version PROPERTIES PASS_REGULAR_EXPRESSION "haolive-server ${PROJECT_VERSION}")
    add_test(NAME licenses COMMAND haolive-server --licenses)
    set_tests_properties(licenses PROPERTIES PASS_REGULAR_EXPRESSION "Useful Sensors")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME bench_script COMMAND ${Python3_EXECUTABLE} -m unittest discover
            -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_bench.py)
    endif()
    if(Python3_Interpreter_FOUND)
        add_test(NAME release_scripts COMMAND ${Python3_EXECUTABLE} -m unittest discover
            -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_release.py)
    endif()
    if(HAOLIVE_BUILD_SHARED)
        add_executable(haolive-model-tests tests/model.cpp)
        target_link_libraries(haolive-model-tests PRIVATE haolive-shared)
        add_executable(haolive-c-header tests/header.c)
        target_link_libraries(haolive-c-header PRIVATE haolive-shared)
        add_test(NAME c_header COMMAND haolive-c-header)
    endif()
endif()

include(GNUInstallDirs)
install(TARGETS haolive-server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(HAOLIVE_BUILD_SHARED)
    install(TARGETS haolive-shared LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/haolive)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/haolive)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/haolive/docs)
install(FILES bench/bench.py DESTINATION ${CMAKE_INSTALL_DATADIR}/haolive/bench)
install(DIRECTORY third_party/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/haolive/licenses)
install(FILES third_party/ggml/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/haolive/licenses RENAME ggml.txt)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
